! DEVICE 3: BR-R1 (Cisco ISR 4321 Edge Router - Site B Remote Branch)
! =====================================================================
enable
configure terminal
hostname BR-R1
no ip domain-lookup
ip domain-name corp.quanumis.lab
service password-encryption

enable algorithm-type scrypt secret StrongEnableSecret#2026!
username netadmin privilege 15 algorithm-type scrypt secret StrongAdminPass#2026!

crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

interface Loopback0
 description BR-R1-RouterID-and-Mgmt
 ip address 10.20.255.1 255.255.255.255
Exit

interface GigabitEthernet0/0/0
 description P2P-WAN-TO-HQ-R1
 ip address 10.255.255.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
Exit

interface GigabitEthernet0/0/1
 description TRANSIT-TO-BR-SW1
 ip address 10.20.0.1 255.255.255.252
 ip ospf network point-to-point
 no shutdown
Exit

router ospf 1
 router-id 10.20.255.1
 passive-interface Loopback0
 network 10.20.0.0 0.0.0.3 area 0
 network 10.20.255.1 0.0.0.0 area 0
 network 10.255.255.0 0.0.0.3 area 0
Exit

ip access-list standard VTY-ADMIN-ALLOW
 deny   10.10.30.0 0.0.0.255
 deny   10.20.30.0 0.0.0.255
 permit 10.10.10.0 0.0.0.255
 permit 10.10.99.0 0.0.0.255
 permit 10.20.10.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny   any log
Exit

line con 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 access-class VTY-ADMIN-ALLOW in
 transport input ssh
 login local
 exec-timeout 5 0
 logging synchronous
Exit
end
write memory


! =====================================================================
! DEVICE 4: BR-SW1 (Cisco Catalyst 3650 Layer 3 Switch - Site B Branch)
! =====================================================================
enable
configure terminal
hostname BR-SW1
no ip domain-lookup
ip domain-name corp.quanumis.lab
service password-encryption
ip routing

enable algorithm-type scrypt secret StrongEnableSecret#2026!
username netadmin privilege 15 algorithm-type scrypt secret StrongAdminPass#2026!

crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

vlan 10
 name CORP_WORKSTATIONS
vlan 20
 name IOT_AND_PRINTERS
vlan 30
 name GUEST_WIFI
vlan 99
 name MGMT_AND_SERVERS
Exit

interface GigabitEthernet1/0/1
 description ROUTED-UPLINK-TO-BR-R1
 no switchport
 ip address 10.20.0.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
Exit

interface range GigabitEthernet1/0/2 - 10
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
Exit

interface range GigabitEthernet1/0/11 - 16
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
Exit

interface range GigabitEthernet1/0/17 - 20
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
Exit

interface range GigabitEthernet1/0/21 - 24
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast
 no shutdown
Exit

ip access-list extended RESTRICT-GUEST-WIFI
 permit udp 10.20.30.0 0.0.0.255 host 203.0.113.1 eq domain
 deny   ip 10.20.30.0 0.0.0.255 10.10.99.0 0.0.0.255 log
 deny   ip 10.20.30.0 0.0.0.255 10.20.99.0 0.0.0.255 log
 deny   ip 10.20.30.0 0.0.0.255 10.0.0.0 0.255.255.255 log
 deny   ip 10.20.30.0 0.0.0.255 172.16.0.0 0.15.255.255 log
 deny   ip 10.20.30.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 permit ip 10.20.30.0 0.0.0.255 any
Exit

interface Vlan10
 ip address 10.20.10.1 255.255.255.0
 no shutdown
Exit

interface Vlan20
 ip address 10.20.20.1 255.255.255.0
 no shutdown
Exit

interface Vlan30
 ip address 10.20.30.1 255.255.255.0
 ip access-group RESTRICT-GUEST-WIFI in
 no shutdown
Exit

interface Vlan99
 ip address 10.20.99.1 255.255.255.0
 no shutdown
Exit

router ospf 1
 router-id 10.20.99.1
 passive-interface default
 no passive-interface GigabitEthernet1/0/1
 network 10.20.0.0 0.0.0.3 area 0
 network 10.20.10.0 0.0.0.255 area 0
 network 10.20.20.0 0.0.0.255 area 0
 network 10.20.30.0 0.0.0.255 area 0
 network 10.20.99.0 0.0.0.255 area 0
Exit

ip access-list standard VTY-ADMIN-ALLOW
 deny   10.10.30.0 0.0.0.255
 deny   10.20.30.0 0.0.0.255
 permit 10.10.10.0 0.0.0.255
 permit 10.10.99.0 0.0.0.255
 permit 10.20.10.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny   any log
Exit

line con 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 access-class VTY-ADMIN-ALLOW in
 transport input ssh
 login local
 exec-timeout 5 0
 logging synchronous
Exit
end
